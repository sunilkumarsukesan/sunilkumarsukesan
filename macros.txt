Sub ValidateAndGenerateFormattedSheet()
    Dim wsSource As Worksheet, wsDest As Worksheet
    Dim lastRow As Long, destRow As Long
    Dim caseDict As Object, stepCount As Object
    Dim i As Long, key As String
    Dim headers As Variant
    Dim userResponse As VbMsgBoxResult

    Set wsSource = ThisWorkbook.Sheets("Sheet1") ' Change if your source sheet is named differently
    Set caseDict = CreateObject("Scripting.Dictionary")
    Set stepCount = CreateObject("Scripting.Dictionary")

    ' Step 1: Validate input
    lastRow = wsSource.Cells(wsSource.Rows.Count, "A").End(xlUp).Row

    For i = 2 To lastRow
        key = Trim(wsSource.Cells(i, 1).Value) ' S.No
        If key <> "" Then
            If Not stepCount.exists(key) Then
                stepCount.Add key, 1
            Else
                stepCount(key) = stepCount(key) + 1
            End If

            If Not caseDict.exists(key) Then
                caseDict.Add key, i
            End If
        End If
    Next i

    ' Check if any test case has no steps
    For Each key In caseDict.keys
        If stepCount(key) = 0 Then
            MsgBox "Test case with S.No '" & key & "' has no steps. Macro aborted.", vbCritical
            Exit Sub
        End If
    Next key

    ' Step 2: Create/Reset FormattedTestCases sheet
    On Error Resume Next
    Set wsDest = ThisWorkbook.Sheets("FormattedTestCases")
    On Error GoTo 0

    If Not wsDest Is Nothing Then
        If Application.WorksheetFunction.CountA(wsDest.Cells) > 0 Then
            userResponse = MsgBox("'FormattedTestCases' sheet already exists and contains data." & vbCrLf & _
                                  "Do you want to clear and recreate it?", vbYesNo + vbQuestion)
            If userResponse = vbNo Then
                MsgBox "Macro cancelled. No changes made.", vbInformation
                Exit Sub
            Else
                wsDest.Cells.Clear
            End If
        End If
    Else
        Set wsDest = ThisWorkbook.Sheets.Add
        wsDest.Name = "FormattedTestCases"
    End If

    headers = Array("Title", "Description", "Area Path", "Iteration Path", "State", "Reason", "Work Item Type", "Assigned To", "Steps")
    For i = 0 To UBound(headers)
        wsDest.Cells(1, i + 1).Value = headers(i)
    Next i

    ' Step 3: Copy test case info (excluding steps)
    destRow = 2
    For Each key In caseDict.keys
        i = caseDict(key)
        wsDest.Cells(destRow, 1).Value = wsSource.Cells(i, 2).Value ' Title
        wsDest.Cells(destRow, 2).Value = wsSource.Cells(i, 3).Value ' Description
        wsDest.Cells(destRow, 3).Value = wsSource.Cells(i, 7).Value ' Area Path
        wsDest.Cells(destRow, 4).Value = wsSource.Cells(i, 8).Value ' Iteration Path
        wsDest.Cells(destRow, 5).Value = wsSource.Cells(i, 9).Value ' State
        wsDest.Cells(destRow, 6).Value = wsSource.Cells(i, 10).Value ' Reason
        wsDest.Cells(destRow, 7).Value = wsSource.Cells(i, 11).Value ' Work Item Type
        wsDest.Cells(destRow, 8).Value = wsSource.Cells(i, 12).Value ' Assigned To
        ' Steps column will be filled in next step
        destRow = destRow + 1
    Next key

    MsgBox "Validation passed and 'FormattedTestCases' sheet generated. Ready for Step 3 (Steps formatting).", vbInformation
End Sub


Function EscapeHTML(str As String) As String
    str = Replace(str, "&", "&amp;")
    str = Replace(str, "<", "&lt;")
    str = Replace(str, ">", "&gt;")
    EscapeHTML = str
End Function

Sub Step3_GenerateStepsXML()
    Dim wsSource As Worksheet, wsDest As Worksheet
    Dim stepsDict As Object
    Dim lastRowSrc As Long, lastRowDest As Long
    Dim rowKey As String, stepsXML As String
    Dim stepList As Collection, stepItem As Variant
    Dim i As Long, currentRow As Long, stepId As Long
    Dim action As String, expected As String

    Set wsSource = ThisWorkbook.Sheets("Sheet1")
    Set wsDest = ThisWorkbook.Sheets("FormattedTestCases")
    Set stepsDict = CreateObject("Scripting.Dictionary")

    lastRowSrc = wsSource.Cells(wsSource.Rows.Count, "E").End(xlUp).Row

    ' STEP 1: Build stepsDict from Sheet1 (grouped by S.No)
    For i = 2 To lastRowSrc
        If Trim(wsSource.Cells(i, 5).Value) <> "" Then ' If Step Description exists
            currentRow = i
            ' Go upwards to find S.No (Column A)
            Do While Trim(wsSource.Cells(currentRow, 1).Value) = "" And currentRow > 1
                currentRow = currentRow - 1
            Loop
            rowKey = Trim(wsSource.Cells(currentRow, 1).Value)

            If rowKey <> "" Then
                If Not stepsDict.exists(rowKey) Then
                    Set stepList = New Collection
                    stepsDict.Add rowKey, stepList
                Else
                    Set stepList = stepsDict(rowKey)
                End If

                action = Trim(wsSource.Cells(i, 5).Value)
                expected = Trim(wsSource.Cells(i, 6).Value)

                If action <> "" And expected <> "" Then
                    stepList.Add Array(action, expected)
                End If
            End If
        End If
    Next i

    ' STEP 2: Inject steps into FormattedTestCases
    lastRowDest = wsDest.Cells(wsDest.Rows.Count, "A").End(xlUp).Row

    For i = 2 To lastRowDest
        rowKey = Trim(wsDest.Cells(i, 10).Value) ' S.No in column J

        If stepsDict.exists(rowKey) Then
            Set stepList = stepsDict(rowKey)
            stepsXML = "<steps id=""0"" last=""" & stepList.Count & """>" & vbCrLf
            stepId = 1

            For Each stepItem In stepList
                stepsXML = stepsXML & "  <step id=""" & stepId & """ type=""ValidateStep"">" & vbCrLf
                stepsXML = stepsXML & "    <parameterizedString isformatted=""true"">" & _
                    EscapeHTML("<DIV><P>" & stepItem(0) & "</P></DIV>") & "</parameterizedString>" & vbCrLf
                stepsXML = stepsXML & "    <parameterizedString isformatted=""true"">" & _
                    EscapeHTML("<DIV><P>" & stepItem(1) & "</P></DIV>") & "</parameterizedString>" & vbCrLf
                stepsXML = stepsXML & "  </step>" & vbCrLf
                stepId = stepId + 1
            Next stepItem

            stepsXML = stepsXML & "</steps>"
            wsDest.Cells(i, 9).Value = stepsXML
        Else
            MsgBox "No steps found for S.No: " & rowKey, vbCritical
            Exit Sub
        End If
    Next i

' ✅ Step 4: Clean up and prepare for paste
    ' Delete S.No (Column J = 10)
    wsDest.Columns(10).Delete

    ' Select and copy the final result table (excluding headers)
    lastRowDest = wsDest.Cells(wsDest.Rows.Count, "A").End(xlUp).Row
    wsDest.Range("A2", wsDest.Cells(lastRowDest, 9)).Copy

    MsgBox "✅ Step 3 complete!" & vbCrLf & _
           "Test cases are now ready to be pasted into the uploader template." & vbCrLf & _
           "(The table has been copied to your clipboard.)", vbInformation
End Sub
